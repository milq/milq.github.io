ondulante.gd
extends Sprite2D  # Avanza ondulando

const MARGEN := 32.0

@export var velocidad := 200.0  # Píxeles por segundo
@export var y_centro := 300.0
@export var amplitud := 50.0    # Píxeles arriba y abajo
@export var frecuencia := 4.0   # Rapidez de la onda

var tiempo := 0.0

func _process(delta: float) -> void:
    tiempo += delta
    position.x += velocidad * delta
    position.y = y_centro + amplitud * sin(tiempo * frecuencia)
    # Al salir por la derecha, vuelve a entrar por la izquierda
    if position.x > get_viewport_rect().size.x + MARGEN:
        position.x = -MARGEN

@export var masa := 1.0              # Más masa, menos agilidad

# Corrige la velocidad con la fuerza de dirección y se mueve
func mover(fuerza: Vector2, delta: float) -> void:
    var aceleracion := fuerza_maxima / masa  # a = F / m
    # Corrige lo que le falta, pero como mucho lo que da de sí
    velocity += fuerza.limit_length(aceleracion * delta)
    velocity = velocity.limit_length(rapidez_maxima)
    move_and_slide()
    if gira and velocity.length() > 10.0:
        rotation = Vector2.UP.angle_to(velocity)

# Buscar: ir hacia un punto a toda velocidad
func buscar(punto: Vector2) -> Vector2:
    var deseada := position.direction_to(punto) * rapidez_maxima
    return deseada - velocity

# Huir: alejarse de un punto, si está a menos de «panico» px
func huir(punto: Vector2, panico := 250.0) -> Vector2:
    if position.distance_to(punto) > panico:
        return -velocity  # Ya está a salvo: frena
    var deseada := punto.direction_to(position) * rapidez_maxima
    return deseada - velocity

# Llegar: buscar, pero frenando dentro del radio de frenado
func llegar(punto: Vector2, frenado := 150.0) -> Vector2:
    var hacia := punto - position
    var factor := minf(hacia.length() / frenado, 1.0)
    var deseada := hacia.normalized() * rapidez_maxima * factor
    return deseada - velocity

# Perseguir: buscar el punto donde estará el otro
func perseguir(otro: CharacterBody2D) -> Vector2:
    var distancia := position.distance_to(otro.position)
    var t := distancia / rapidez_maxima  # Segundos
    return buscar(otro.position + otro.velocity * t)

# Evadir: huir del punto donde estará el perseguidor
func evadir(otro: CharacterBody2D, panico := 250.0) -> Vector2:
    var distancia := position.distance_to(otro.position)
    var t := distancia / rapidez_maxima
    return huir(otro.position + otro.velocity * t, panico)

var angulo_vagar := randf() * TAU  # Dónde está el punto

# Vagar: buscar un punto de un círculo que va por delante
func vagar(delta: float) -> Vector2:

    var centro := position + velocity.normalized() * 100.0
    var punto := centro + Vector2.from_angle(angulo_vagar) * 60

func evitar(rayo: RayCast2D) -> Vector2:
    var delante := velocity.normalized()
    var largo := 40.0 + velocity.length() * 0.5  # Más si corre
    var punta := global_position + delante * largo
    rayo.target_position = rayo.to_local(punta)
    rayo.force_raycast_update()
    if not rayo.is_colliding():
        return Vector2.ZERO
    # Gira hacia el lado al que apunta la normal del obstáculo
    var lado := delante.orthogonal()
    if lado.dot(rayo.get_collision_normal()) < 0.0:
        lado = -lado
    return lado * fuerza_maxima

var punto_actual := 0  # Índice del punto del camino al que va

func separar(vecinos: Array[Node], radio := 60.0) -> Vector2:
    var lejos := Vector2.ZERO
    for nodo in vecinos:
        var otro := nodo as Node2D
        var desde := position - otro.position
        if otro != self and desde.length() < radio:
            # Cuanto más cerca está el vecino, más pesa
            var cerca := radio - desde.length()
            lejos += desde.normalized() * cerca
    if lejos == Vector2.ZERO:
        return Vector2.ZERO
    return lejos.normalized() * rapidez_maxima - velocity

    var suma := Vector2.ZERO
    for nodo in vecinos:
        var otro := nodo as CharacterBody2D
        var distancia := position.distance_to(otro.position)
        if otro != self and distancia < radio:
            suma += otro.velocity
    if suma == Vector2.ZERO:
        return Vector2.ZERO
    return suma.normalized() * rapidez_maxima - velocity

    var centro := Vector2.ZERO
    var cuantos := 0

            centro += otro.position
            cuantos += 1
    if cuantos == 0:

func seguir(lider: CharacterBody2D, hueco := 60.0) -> Vector2:
    if lider.velocity.length() < 10.0:
        return llegar(position)  # Quiere estar quieto: frena
    var detras := -lider.velocity.normalized() * hueco
    return llegar(lider.position + detras)

# A*: el camino más barato de inicio a meta
func a_estrella(grafo: Dictionary, inicio: Vector2i,
        meta: Vector2i) -> Array[Vector2i]:
    var frontera := {inicio: 0.0}  # Casilla → f, por explorar
    var coste := {inicio: 0.0}     # Casilla → g, el mejor
    var viene_de := {inicio: inicio}
    while not frontera.is_empty():
        # La casilla de la frontera con menor f
        var actual: Vector2i = frontera.keys()[0]
        for casilla in frontera:
            if frontera[casilla] < frontera[actual]:
                actual = casilla
        frontera.erase(actual)
        if actual == meta:
            break
        for vecina in grafo[actual]:
            var g: float = coste[actual] + grafo[actual][vecina]
            if not coste.has(vecina) or g < coste[vecina]:
                coste[vecina] = g  # Un camino mejor hasta ella
                viene_de[vecina] = actual
                frontera[vecina] = g + estimar(vecina, meta)
    return reconstruir(viene_de, inicio, meta)

# Heurística: la distancia Manhattan hasta la meta
func estimar(casilla: Vector2i, meta: Vector2i) -> float:
    return absi(casilla.x - meta.x) + absi(casilla.y - meta.y)

# Sigue viene_de desde la meta hasta el inicio
func reconstruir(viene_de: Dictionary, inicio: Vector2i,
        meta: Vector2i) -> Array[Vector2i]:
    var camino: Array[Vector2i] = []
    if not viene_de.has(meta):
        return camino  # No hay camino: vacío
    var casilla := meta
    while casilla != inicio:
        camino.push_front(casilla)
        casilla = viene_de[casilla]
    camino.push_front(inicio)
    return camino

static func generar(columnas: int, filas: int,
        desde: Vector2i) -> Array[String]:
    var mapa: Array[String] = []
    for fila in filas:
        mapa.append("#".repeat(columnas))  # Todo es muro

    Vector2i.UP, Vector2i.RIGHT, Vector2i.DOWN, Vector2i.LEFT]
const ALFA := 0.5     # Tasa de aprendizaje
const GAMMA := 0.9    # Factor de descuento
const EPSILON := 0.2  # Probabilidad de explorar

var q: Dictionary[Vector2i, Array] = {}  # Casilla → 4 valores

func _ready() -> void:
    seed(7)  # El mismo azar en cada ejecución
    for fila in MAPA.size():
        for columna in MAPA[fila].length():
            q[Vector2i(columna, fila)] = [0.0, 0.0, 0.0, 0.0]
    for episodio in 300:
        var total := episodio_desde(Vector2i(0, 2))
        print("Episodio ", episodio, ": ", total)
    mostrar()

# Un episodio, desde una casilla hasta el queso o la trampa
func episodio_desde(s: Vector2i) -> float:
    var total := 0.0  # Recompensa acumulada
    while letra(s) != "Q" and letra(s) != "T":
        var a := elegir(s)
        var s2 := mover(s, a)
        var r := recompensa(s2)
        var mejor: float = q[s2].max()  # Lo que espera allí
        q[s][a] += ALFA * (r + GAMMA * mejor - q[s][a])
        total += r
        s = s2
    return total

func elegir(s: Vector2i) -> int:
    if randf() < EPSILON:
        return randi() % ACCIONES.size()  # Explora
    var valores: Array = q[s]
    return valores.find(valores.max())  # Explota

func mover(s: Vector2i, a: int) -> Vector2i:
    var destino := s + ACCIONES[a]
    if not q.has(destino) or letra(destino) == "#":
        return s  # Fuera del mapa o contra el muro: no se mueve
    return destino

func recompensa(s: Vector2i) -> float:
    match letra(s):
        "Q":
            return 10.0   # El queso
        "T":
            return -10.0  # La trampa
    return -1.0  # Cada paso cuesta

func letra(s: Vector2i) -> String:
    return MAPA[s.y][s.x]

func mostrar() -> void:
    const FLECHAS := ["↑", "→", "↓", "←"]
    for fila in MAPA.size():
        var linea := ""
        for columna in MAPA[fila].length():
            var v: Array = q[Vector2i(columna, fila)]
            var i: int = v.find(v.max())  # La mejor acción
            linea += "%s %5.2f   " % [FLECHAS[i], v[i]]
        print(linea)

# El camino de Niblo con lo aprendido, sin explorar
func camino_aprendido(s: Vector2i) -> Array[Vector2i]:
    var camino: Array[Vector2i] = [s]
    for _paso in 20:  # Como mucho 20 pasos, por si no aprendió
        if letra(s) == "Q" or letra(s) == "T":
            break
        var v: Array = q[s]
        s = mover(s, v.find(v.max()))  # Siempre la mejor acción
        camino.append(s)
    return camino
