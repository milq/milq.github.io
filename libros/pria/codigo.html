Capítulo 4. Trigonometría e interpolación

Un lanzapelotas con vida

Con el ratón como jugador, el lanzapelotas es este script:

extends Sprite2D  # El lanzapelotas

const GIRO := deg_to_rad(90.0)  # Por segundo
const CONO := deg_to_rad(45.0)  # A cada lado
const APLASTADA := Vector2(0.7, 1.3)
var tiempo := 0.0
var tw: Tween  # El tween del último lanzamiento

func _process(delta: float) -> void:
    var jugador := get_global_mouse_position()
    var hacia := jugador - position
    var meta := hacia.angle()  # Al jugador...
    if absf(transform.x.angle_to(hacia)) > CONO:
        tiempo += delta  # ...o barre ±45°
        meta = PI / 4 * sin(TAU * tiempo / 4.0)
    rotation = rotate_toward(
            rotation, meta, GIRO * delta)
    if tw and tw.is_running():
        return  # Aún está lanzando
    var desvio := transform.x.angle_to(hacia)
    if is_zero_approx(desvio):
        lanzar()

func lanzar() -> void:  # Anima el disparo
    tw = create_tween()
    tw.tween_property(
            self, "scale", APLASTADA, 0.05)
    tw.set_trans(Tween.TRANS_BACK)  # Se pasa
    tw.tween_property(
            self, "scale", Vector2.ONE, 0.3)
    tw.tween_interval(0.5)          # Recarga

extends CharacterBody2D  # El jugador

const RAPIDEZ := 300.0       # px/s
const ACELERACION := 1500.0  # px/s²
const SALTO := 490.0         # px/s hacia arriba

func _physics_process(delta: float) -> void:
    var eje := Input.get_axis(
            "ui_left", "ui_right")
    var meta := eje * RAPIDEZ        # ±300 o 0

    velocity.x = move_toward(
            velocity.x, meta, paso)
    if is_on_floor():
        $Coyote.start(0.1)  # Mientras pisa
    else:
        velocity += get_gravity() * delta
    if Input.is_action_just_pressed(
            "ui_accept"):

    var puede: bool = not $Coyote.is_stopped()
    if puede and not $Bufer.is_stopped():
        velocity.y = -SALTO
        $Coyote.stop()
        $Bufer.stop()

            "ui_accept"):

            velocity.y /= 2.0  # salto variable

Capítulo 8. Steering behaviors

El vehículo

En Godot, en 2D, el vehículo es un CharacterBody2D con el Motion Mode en Floating, y el método limit_length() acorta un vector si mide más que el número que recibe. Con un Sprite2D hijo, este script lleva el monstruo hacia el ratón:

extends CharacterBody2D  # Un vehículo

@export var rapidez_maxima := 250.0  # px/s
@export var fuerza_maxima := 600.0
@export var masa := 1.0

func _physics_process(delta: float) -> void:
    var raton := get_global_mouse_position()
    mover(buscar(raton), delta)

func mover(
        fuerza: Vector2, delta: float) -> void:
    var paso := fuerza_maxima / masa * delta
    velocity += fuerza.limit_length(paso)
    velocity = velocity.limit_length(
            rapidez_maxima)
    move_and_slide()
    rotation = velocity.angle()

func buscar(punto: Vector2) -> Vector2:
    var hacia := position.direction_to(punto)
    return hacia * rapidez_maxima - velocity

Vagar

En Godot, en 2D, la memoria vive fuera de la función:

var angulo := 0.0  # Del punto; se conserva

func vagar(delta: float) -> Vector2:
    angulo += randf_range(-10.0, 10.0) * delta
    var centro := velocity.normalized() * 85.0
    var direccion := Vector2.from_angle(angulo)
    var punto := direccion * 60.0
    return buscar(position + centro + punto)

Bandadas

En Godot, cada monstruo se une al grupo en _ready() con add_to_group("grupo"), y bandada() lo recorre:

func bandada() -> Vector2:  # Vecinos a < 100 px
    var lejos := Vector2.ZERO   # Separación
    var media := Vector2.ZERO   # Velocidades
    var centro := Vector2.ZERO  # Posiciones
    var n: int = 0              # Vecinos
    for otro in get_tree().get_nodes_in_group(
            "grupo"):
        var punto: Vector2 = otro.position
        var desde := position - punto
        var cerca := desde.length() < 100.0
        if otro != self and cerca:
            lejos += (desde.normalized()
                    * 100.0 - desde)
            media += otro.velocity
            centro += otro.position
            n += 1
    if n == 0: return Vector2.ZERO
    var alinear := media / n - velocity
    var juntar := buscar(centro / n)
    return lejos * 2.0 + alinear + juntar * 0.5

Un grupo con prioridades

Los seis deben pasear juntos y salir a por el jugador a menos de 300 píxeles, pero si se suman las fuerzas, la cohesión los frena. La prioridad es una forma de combinar comportamientos que usa solo el más urgente de los que piden algo:

@export var jugador: CharacterBody2D

func _physics_process(delta: float) -> void:
    var d := position.distance_to(
            jugador.position)
    if d < 300.0:  # 1.º: ir a por el jugador
        mover(perseguir(jugador), delta)
    else:          # 2.º: ir con el grupo
        mover(bandada() + vagar(delta) * 0.3,
                delta)  # A 1, se rompe el grupo

Un monstruo que persigue al jugador

En un Sprite2D con textura y un Timer hijo, con Wait Time a 0.5 y Autostart activado, este script persigue al ratón:

extends Sprite2D  # Persigue al ratón

const RAPIDEZ := 200.0  # Píxeles por segundo
var red := AStarGrid2D.new()  # Grafo del nivel
var camino: PackedVector2Array = []  # Waypoints

func _ready() -> void:
    red.region = Rect2i(0, 0, 9, 6)
    red.cell_size = Vector2(100, 100)  # Tamaño
    red.offset = Vector2(50, 50)       # Centro
    var modo := AStarGrid2D.DIAGONAL_MODE_NEVER
    red.diagonal_mode = modo
    var h := AStarGrid2D.HEURISTIC_MANHATTAN
    red.default_estimate_heuristic = h
    red.update()
    for y in range(4):
        red.set_point_solid(Vector2i(4, y))
    var barro := Vector2i(4, 4)
    red.set_point_weight_scale(barro, 5.0)
    position = Vector2(150, 250)
    $Timer.timeout.connect(buscar)

func buscar() -> void:
    var desde := Vector2i(
            (position / 100.0).floor())
    var raton := get_global_mouse_position()
    var meta := Vector2i(
            (raton / 100.0).floor())
    if red.is_in_boundsv(meta):
        camino = red.get_point_path(desde, meta)

func _process(delta: float) -> void:
    if camino.is_empty():
        return
    position = position.move_toward(
            camino[0], RAPIDEZ * delta)
    if position == camino[0]:  # Llega: lo quita
        camino.remove_at(0)

Capítulo 11. Máquinas de estados

Estados con enum y match

En Godot, en 2D, un Sprite2D con el icono de Godot y este script tiene el esqueleto de la máquina:

extends Sprite2D  # El monstruo, con su máquina

enum Estado {
        PATRULLAR, PERSEGUIR, ATRAPAR, BUSCAR }

@export var jugador: Node2D
var estado := Estado.PATRULLAR

func _physics_process(delta: float) -> void:
    var d := position.distance_to(
            jugador.position)
    match estado:
        Estado.PATRULLAR:
            patrullar(d, delta)
        Estado.PERSEGUIR:
            perseguir(d, delta)
        Estado.ATRAPAR:
            atrapar(d)
        Estado.BUSCAR:
            buscar(d, delta)
        _:
            push_error("Estado inesperado")

Transiciones

func cambiar_a(nuevo: Estado) -> void:

Un monstruo que vigila

El resto del script son las constantes y variables, que van junto a estado, y las funciones de los cuatro estados:

const VISTA := 200.0   # Píxeles a los que lo ve
const ALCANCE := 40.0  # Píxeles para atraparlo
const RUTA: Array[Vector2] = [
        Vector2(100, 200), Vector2(900, 200)]
var siguiente: int = 1     # Punto al que va
var visto := Vector2.ZERO  # Dónde lo vio
var espera := 0.0          # Segundos buscando

func patrullar(d: float, delta: float) -> void:
    var meta := RUTA[siguiente]
    position = position.move_toward(
            meta, 100 * delta)
    if position == meta:  # Llega: al otro punto
        var puntos := RUTA.size()
        siguiente = (siguiente + 1) % puntos
    if d < VISTA:
        cambiar_a(Estado.PERSEGUIR)

func perseguir(d: float, delta: float) -> void:
    visto = jugador.position
    position = position.move_toward(
            visto, 160 * delta)
    if d <= ALCANCE:
        cambiar_a(Estado.ATRAPAR)
    elif d > VISTA:
        cambiar_a(Estado.BUSCAR)

func atrapar(d: float) -> void:
    if d > ALCANCE:
        cambiar_a(Estado.PERSEGUIR)

func buscar(d: float, delta: float) -> void:
    position = position.move_toward(
            visto, 160 * delta)
    espera += delta
    if d < VISTA:
        cambiar_a(Estado.PERSEGUIR)
    elif espera > 6.0:
        cambiar_a(Estado.PATRULLAR)

extends Node  # Un laberinto perfecto de 13 × 9

const SALTOS: Array[Vector2i] = [
        Vector2i(2, 0), Vector2i(-2, 0),
        Vector2i(0, 2), Vector2i(0, -2)]

func _ready() -> void:
    seed(2026)
    var mapa: Array[String] = []
    for fila in 9:
        mapa.append("#".repeat(13))  # Todo muro
    var zona := Rect2i(1, 1, 11, 7)  # Sin borde
    var pila: Array[Vector2i] = [Vector2i(1, 1)]
    mapa[1][1] = "."
    while not pila.is_empty():
        var actual: Vector2i = pila.back()
        var cerradas: Array[Vector2i] = []
        for salto in SALTOS:
            var c := actual + salto
            var dentro := zona.has_point(c)
            if dentro and mapa[c.y][c.x] == "#":
                cerradas.append(c)
        if cerradas.is_empty():
            pila.pop_back()  # Vuelve atrás
            continue
        var nueva: Vector2i = (
                cerradas.pick_random())
        var muro := (actual + nueva) / 2
        mapa[muro.y][muro.x] = "."
        mapa[nueva.y][nueva.x] = "."
        pila.push_back(nueva)
    for fila in mapa:
        print(fila)

class_name Asteroide
extends Area2D  # Persigue o hace eses

enum Modo { PERSEGUIR, ZIGZAG }

@export var modo := Modo.PERSEGUIR
@export var rapidez := 140.0  # px/s
@export var giro := 2.0       # rad/s
@export var objetivo: Node2D  # La nave

var tiempo := 0.0

func _physics_process(delta: float) -> void:
    tiempo += delta
    match modo:
        Modo.PERSEGUIR: perseguir(delta)
        Modo.ZIGZAG: zigzag(delta)

# Con rotación: gira hacia la nave y avanza
func perseguir(delta: float) -> void:
    var hacia := objetivo.position - position
    var meta := Vector2.UP.angle_to(hacia)
    var paso := giro * delta
    rotation = rotate_toward(
            rotation, meta, paso)
    var delante := Vector2.UP.rotated(rotation)
    position += delante * rapidez * delta

# Sin rotación: va hacia la nave haciendo eses
func zigzag(delta: float) -> void:
    var nave := objetivo.position
    var hacia := position.direction_to(nave)
    var onda := sin(tiempo * 5.0)
    var lado := hacia.orthogonal() * onda
    mover_hacia(position + hacia + lado, delta)

func mover_hacia(
        punto: Vector2, delta: float) -> void:
    var hacia := position.direction_to(punto)
    position += hacia * rapidez * delta
    if hacia != Vector2.ZERO:
        rotation = Vector2.UP.angle_to(hacia)

class_name Vehiculo
extends CharacterBody2D  # Se mueve con fuerzas

@export var rapidez_maxima := 250.0  # px/s
@export var fuerza_maxima := 600.0
@export var masa := 1.0   # Más masa, menos ágil
@export var gira := true  # Mira hacia donde va

func mover(
        fuerza: Vector2, delta: float) -> void:
    var paso := fuerza_maxima / masa * delta
    velocity += fuerza.limit_length(paso)
    velocity = velocity.limit_length(
            rapidez_maxima)
    move_and_slide()
    if gira and velocity.length() > 10.0:
        rotation = Vector2.UP.angle_to(velocity)

func buscar(punto: Vector2) -> Vector2:
    var hacia := position.direction_to(punto)
    return hacia * rapidez_maxima - velocity

func huir(punto: Vector2,
        panico: float) -> Vector2:
    if position.distance_to(punto) > panico:
        return -velocity  # A salvo: frena
    var lejos := punto.direction_to(position)
    return lejos * rapidez_maxima - velocity

func llegar(punto: Vector2,
        frenado: float) -> Vector2:
    var distancia := position.distance_to(punto)
    var factor := minf(distancia / frenado, 1.0)
    var hacia := position.direction_to(punto)
    var deseada := hacia * rapidez_maxima
    return deseada * factor - velocity

            var c := actual + salto

        if pila.size() > profundidad:
