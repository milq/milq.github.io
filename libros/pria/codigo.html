Un lanzapelotas con vida

En un Sprite2D con el icono, en (200, 100), y el ratón como jugador, este script es el lanzapelotas:

extends Sprite2D  # El lanzapelotas que vigila y lanza

const GIRO := deg_to_rad(90.0)  # Velocidad angular
const CONO := deg_to_rad(20.0)  # A cada lado
const APLASTADA := Vector2(0.7, 1.3)
var tiempo := 0.0
var tw: Tween  # El tween del último lanzamiento

func _process(delta: float) -> void:
    var hacia := get_global_mouse_position() - position
    var meta := hacia.angle()  # Apunta al jugador...
    if absf(transform.x.angle_to(hacia)) > CONO:
        tiempo += delta  # ...o barre ±45° cada 4 s
        meta = PI / 4 * sin(TAU * tiempo / 4.0)
    var paso := GIRO * delta  # Lo que gira, como mucho
    rotation = rotate_toward(rotation, meta, paso)
    if tw and tw.is_running():
        return  # Aún está lanzando
    if is_zero_approx(transform.x.angle_to(hacia)):
        lanzar()

func lanzar() -> void:
    tw = create_tween()
    tw.tween_property(self, "scale", APLASTADA, 0.05)
    tw.set_trans(Tween.TRANS_BACK)  # Se pasa y vuelve
    tw.tween_property(self, "scale", Vector2.ONE, 0.3)
    tw.tween_interval(0.5)          # Recarga

extends CharacterBody2D  # El jugador: corre y salta

const RAPIDEZ := 300.0       # px/s
const ACELERACION := 1500.0  # px/s²
const SALTO := 490.0         # px/s hacia arriba

func _physics_process(delta: float) -> void:
    var eje := Input.get_axis("ui_left", "ui_right")
    var meta := eje * RAPIDEZ  # −300, 0 o 300
    var paso := ACELERACION * delta  # 25 px/s
    velocity.x = move_toward(velocity.x, meta, paso)
    if is_on_floor():
        $Coyote.start(0.1)  # Se renueva mientras pisa
    else:
        velocity += get_gravity() * delta
    if Input.is_action_just_pressed("ui_accept"):
        $Bufer.start(0.1)   # Recuerda la pulsación
    var puede: bool = not $Coyote.is_stopped()
    if puede and not $Bufer.is_stopped():
        velocity.y = -SALTO
        $Coyote.stop()
        $Bufer.stop()
    if Input.is_action_just_released("ui_accept"):
        if velocity.y < 0.0:   # Si aún sube,
            velocity.y /= 2.0  # salto variable
    move_and_slide()

    [Vector2(-200, 0), 2.0],  # Izquierda

Capítulo 8. Steering behaviors

El vehículo

Con ese nodo en (200, 100) y un Sprite2D hijo con el icono de Godot, este script lleva el monstruo hacia el ratón:

extends CharacterBody2D  # El monstruo, un vehículo

@export var rapidez_maxima := 250.0  # px/s
@export var fuerza_maxima := 600.0
@export var masa := 1.0

func _physics_process(delta: float) -> void:
    var raton := get_global_mouse_position()
    mover(buscar(raton), delta)

func mover(fuerza: Vector2, delta: float) -> void:
    var paso := fuerza_maxima / masa * delta
    velocity += fuerza.limit_length(paso)
    velocity = velocity.limit_length(rapidez_maxima)
    move_and_slide()
    rotation = velocity.angle()  # Hacia donde va

func buscar(punto: Vector2) -> Vector2:
    var hacia := position.direction_to(punto)
    return hacia * rapidez_maxima - velocity

Capítulo 9. Tiles, waypoints y pathfinding

Un monstruo que persigue al jugador

En un Sprite2D con textura y un Timer hijo, con Wait Time a 0,5 y Autostart activado, este script hace que el monstruo persiga al ratón:

extends Sprite2D  # El monstruo persigue al ratón

const RAPIDEZ := 200.0  # Píxeles por segundo
var red := AStarGrid2D.new()  # El nivel, como grafo
var camino: PackedVector2Array = []  # Sus waypoints

func _ready() -> void:
    red.region = Rect2i(0, 0, 9, 6)    # Las casillas
    red.cell_size = Vector2(100, 100)  # Su tamaño
    red.offset = Vector2(50, 50)       # Al centro
    red.diagonal_mode = AStarGrid2D.DIAGONAL_MODE_NEVER
    red.update()
    for y in range(4):
        red.set_point_solid(Vector2i(4, y))  # El muro
    red.set_point_weight_scale(Vector2i(4, 4), 5.0)
    position = Vector2(250, 150)
    $Timer.timeout.connect(buscar)

func buscar() -> void:
    var desde := Vector2i((position / 100.0).floor())
    var raton := get_global_mouse_position() / 100.0
    var hasta := Vector2i(raton.floor())
    if red.is_in_boundsv(hasta):
        camino = red.get_point_path(desde, hasta)

func _process(delta: float) -> void:
    if camino.is_empty():
        return
    var paso := RAPIDEZ * delta
    position = position.move_toward(camino[0], paso)
    if position == camino[0]:  # Llega: quita el punto
        camino.remove_at(0)

Capítulo 11. Máquinas de estados

Estados con enum y match

En Godot, en 2D, un Sprite2D con el icono de Godot y este script tiene el esqueleto de la máquina:

extends Sprite2D  # El monstruo, con su máquina

enum Estado { PATRULLAR, PERSEGUIR, ATRAPAR, BUSCAR }

@export var jugador: Node2D
var estado := Estado.PATRULLAR

func _physics_process(delta: float) -> void:
    var d := position.distance_to(jugador.position)
    match estado:
        Estado.PATRULLAR:
            patrullar(d, delta)
        Estado.PERSEGUIR:
            perseguir(d, delta)
        Estado.ATRAPAR:
            atrapar(d)
        Estado.BUSCAR:
            buscar(d, delta)
        _:
            push_error("Estado inesperado")

Transiciones

En Godot, lo más cómodo es que una función propia, cambiar_a(), sea el único sitio del script que cambia el estado:

func cambiar_a(nuevo: Estado) -> void:
    estado = nuevo
    espera = 0.0                           # Entrada
    $Etiqueta.text = Estado.keys()[nuevo]  # Su nombre

Un monstruo que vigila

El resto del script son las constantes y las variables, que van arriba, junto a estado, y las funciones de los cuatro estados, con sus transiciones al final:

const VISTA := 200.0   # Píxeles a los que lo ve
const ALCANCE := 40.0  # Píxeles a los que atrapa
const RUTA: Array[Vector2] = [
        Vector2(200, 100), Vector2(1000, 100)]
var siguiente: int = 1     # Punto al que va
var visto := Vector2.ZERO  # Dónde lo vio la última vez
var espera := 0.0          # Segundos buscando

func patrullar(d: float, delta: float) -> void:
    var meta := RUTA[siguiente]
    position = position.move_toward(meta, 100 * delta)
    if position == meta:  # Ha llegado: al otro punto
        siguiente = (siguiente + 1) % RUTA.size()
    if d < VISTA:
        cambiar_a(Estado.PERSEGUIR)

func perseguir(d: float, delta: float) -> void:
    visto = jugador.position
    position = position.move_toward(visto, 160 * delta)
    if d <= ALCANCE:
        cambiar_a(Estado.ATRAPAR)
    elif d > VISTA:
        cambiar_a(Estado.BUSCAR)

func atrapar(d: float) -> void:
    if d > ALCANCE:
        cambiar_a(Estado.PERSEGUIR)

func buscar(d: float, delta: float) -> void:
    position = position.move_toward(visto, 160 * delta)
    espera += delta
    if d < VISTA:
        cambiar_a(Estado.PERSEGUIR)
    elif espera > 6.0:
        cambiar_a(Estado.PATRULLAR)

    velocity += fuerza.limit_length(paso)
    velocity = velocity.limit_length(rapidez_maxima)
    move_and_slide()
    if gira and velocity.length() > 10.0:

    return hacia * rapidez_maxima - velocity

func huir(punto: Vector2, panico: float) -> Vector2:

    return lejos * rapidez_maxima - velocity

func llegar(punto: Vector2, frenado: float) -> Vector2:
    var distancia := position.distance_to(punto)
    var factor := minf(distancia / frenado, 1.0)
    var hacia := position.direction_to(punto)
    return hacia * rapidez_maxima * factor - velocity

func separar(vecinos: Array[Node]) -> Vector2:
    var lejos := Vector2.ZERO
    for otro in vecinos:
        var desde: Vector2 = position - otro.position
        var d := desde.length()
        if otro != self and d < 60.0:
            lejos += desde.normalized() * (60.0 - d)
    if lejos == Vector2.ZERO:
        return Vector2.ZERO
    return buscar(position + lejos)

static func generar(columnas: int, filas: int,
        desde: Vector2i) -> Array[String]:
    var mapa: Array[String] = []
    for fila in filas:
        mapa.append("#".repeat(columnas))  # Todo, muro
    var zona := Rect2i(1, 1, columnas - 2, filas - 2)
    var pila: Array[Vector2i] = [desde]
    var salida := desde
    var profundidad: int = 1
    mapa[desde.y][desde.x] = "."
    while not pila.is_empty():
        var actual: Vector2i = pila.back()
        var cerradas: Array[Vector2i] = []
        for salto in SALTOS:
            var c := actual + salto
            var dentro := zona.has_point(c)
            if dentro and mapa[c.y][c.x] == "#":
                cerradas.append(c)
        if cerradas.is_empty():
            pila.pop_back()  # Callejón: vuelve atrás
            continue
        var nueva: Vector2i = cerradas.pick_random()
        var muro := (actual + nueva) / 2  # En medio
        mapa[muro.y][muro.x] = "."
        mapa[nueva.y][nueva.x] = "."
        pila.push_back(nueva)
        if pila.size() > profundidad:  # La más lejana
            profundidad = pila.size()
            salida = nueva
    mapa[salida.y][salida.x] = "S"
    return mapa
